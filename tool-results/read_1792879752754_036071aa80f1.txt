     1→/**
     2→ * MCP (Model Context Protocol) tool layer for VisionCut AI.
     3→ *
     4→ * Exposes the editor's AI capabilities as MCP tools so Claude (Desktop, Code,
     5→ * or claude.ai connectors) can drive them remotely. The wire protocol lives in
     6→ * src/app/api/mcp/route.ts (JSON-RPC 2.0 over Streamable HTTP, stateless).
     7→ *
     8→ * Auth: MCP_TOKEN env — accepted as "Authorization: Bearer <token>" or
     9→ * "?key=<token>" on the URL (the URL form is what Claude Desktop custom
    10→ * connectors need, since they send no headers).
    11→ *
    12→ * Tools reuse the same provider-agnostic engines as the app itself:
    13→ *   llm.ts (storyboard) · image.ts · voiceover.ts · video.ts · stickers.ts · db.ts
    14→ */

   274→function splitDataUrl(dataUrl: string): { mimeType: string; base64: string } {
   275→  const match = /^data:([^;]+);base64,([\s\S]*)$/.exec(dataUrl);
   276→  if (!match) return { mimeType: "application/octet-stream", base64: "" };
   277→  return { mimeType: match[1], base64: match[2] };
   278→}
   279→
   280→function textResult(text: string, isError = false): McpToolResult {
   281→  return { content: [{ type: "text", text }], ...(isError ? { isError: true } : {}) };
   282→}
   283→
   284→function fail(err: unknown, label: string): McpToolResult {
   285→  if (err instanceof ImageError || err instanceof VoiceoverError || err instanceof VideoError) {
   286→    return textResult(`${label} failed: ${err.message}`, true);
   287→  }
   288→  const msg = err instanceof Error ? err.message : String(err);
   289→  return textResult(`${label} failed: ${msg}`, true);
   290→}

   296→async function toolStatus(): Promise<McpToolResult> {
   297→  let projectCount = 0;
   298→  try {
   299→    projectCount = await db.project.count();
   300→  } catch {
   301→    /* db optional for status */
   302→  }
   303→  const balance = await getWavespeedBalance().catch(() => null);
   304→  const balText = balance === null ? "unknown" : `$${balance.toFixed(2)}${balance < 0.05 ? " — LOW, video generation will stop soon" : ""}`;
   305→  const kieCredit = await getKieCredit().catch(() => null);
   306→  const kieText = kieCredit === null ? "no key" : `key ok, credits ${kieCredit}`;
   307→  const lines = [
   308→    "VisionCut AI — engine status",
   309→    `- script/LLM: provider=${llmDiagnostics().provider}, custom key=${llmDiagnostics().custom.hasKey}, model=${llmDiagnostics().custom.model}`,
   310→    `- image: provider=${imageDiagnostics().provider}, wavespeed key=${imageDiagnostics().keys.wavespeed}, model=${imageDiagnostics().models.wavespeed}`,
   311→    `- voiceover: provider=${voiceoverDiagnostics().provider}, wavespeed key=${voiceoverDiagnostics().keys.wavespeed}, fast=${voiceoverDiagnostics().models.wavespeedFast}, hd=${voiceoverDiagnostics().models.wavespeedHd}`,

   313→    `- free fallbacks (auto-engage when WaveSpeed fails or is out of credits): image→pollinations.ai, voiceover→google-translate-tts. Video has NO free fallback.`,

   316→    `- projects in database: ${projectCount}`,

   321→async function toolListProjects(args: z.infer<typeof ListProjectsArgs>): Promise<McpToolResult> {
   322→  const limit = args?.limit ?? 20;
   323→  const demo = await db.user.findFirst({ where: { isDemo: true }, select: { id: true } });
   324→  const projects = await db.project.findMany({
   325→    where: demo ? { userId: demo.id } : {},
   326→    orderBy: { updatedAt: "desc" },
   327→    take: limit,
   328→    select: { id: true, name: true, aspectRatio: true, data: true, updatedAt: true },
   329→  });
   330→  if (!projects.length) return textResult("No projects yet. Open the editor and create one.");
   331→  const lines = projects.map((p) => {
   332→    let tracks = 0;
   333→    let clips = 0;
   334→    try {
   335→      const data = JSON.parse(p.data || "{}") as { tracks?: Array<{ clips?: unknown[] }> };
   336→      tracks = Array.isArray(data.tracks) ? data.tracks.length : 0;
   337→      clips = (data.tracks || []).reduce((n, t) => n + (Array.isArray(t.clips) ? t.clips.length : 0), 0);
   338→    } catch {
   339→      /* unparseable data — skip counts */
   340→    }
   341→    return `- ${p.name} (id=${p.id}, ${p.aspectRatio}, ${tracks} tracks / ${clips} clips, updated ${p.updatedAt.toISOString()})`;
   342→  });
   343→  return textResult(`Projects (${projects.length}):\n${lines.join("\n")}`);
   344→}

   346→async function toolSearchStickers(raw: unknown): Promise<McpToolResult> {
   347→  const args = SearchStickersArgs.parse(raw);
   348→  const hits = searchStickers(args.query, { group: args.group, limit: args.limit ?? 12 });
   349→  if (!hits.length) return textResult(`No stickers found for "${args.query}". Try another word or an emoji.`);
   350→  const lines = hits.map(
   351→    (h, i) => `${i + 1}. ${h.emoji || "•"} ${h.annotation} [${h.source}${h.group ? `, ${h.group}` : ""}]\n   ${h.url}`,
   352→  );
   353→  return textResult(`Stickers for "${args.query}" (${hits.length}):\n${lines.join("\n")}`);
   354→}

   356→async function toolStoryboard(raw: unknown): Promise<McpToolResult> {
   357→  const args = StoryboardArgs.parse(raw);
   358→  const completion = await generateChatCompletion(
   359→    [
   360→      { role: "system", content: STORYBOARD_SYSTEM },
   361→      {
   362→        role: "user",
   363→        content: `Convert this script into a scene-by-scene storyboard.\nVisual style: ${args.style || "cinematic"}\nAspect ratio: ${args.aspect_ratio || "16:9"}\n\nSCRIPT:\n${args.script}\n\n${STORYBOARD_SCHEMA_INSTRUCTION}`,
   364→      },
   365→    ],
   366→    { temperature: 0.7 },
   367→  );
   368→  const storyboard = extractJsonObject(completion.content) as
   369→    | { scenes?: unknown; title?: string; musicSuggestion?: string }
   370→    | null;
   371→  if (!storyboard || !Array.isArray(storyboard.scenes) || storyboard.scenes.length === 0) {
   372→    return textResult("The AI returned an unreadable storyboard. Please rephrase the script and retry.", true);
   373→  }
   374→  return textResult(
   375→    `Storyboard "${storyboard.title || "Untitled"}" — ${storyboard.scenes.length} scenes (provider=${completion.provider}, model=${completion.model})\n\n` +
   376→      JSON.stringify(storyboard, null, 2),
   377→  );
   378→}

   431→async function toolCheckVideo(raw: unknown): Promise<McpToolResult> {
   432→  const args = CheckVideoArgs.parse(raw);
   433→  const result = await pollVideoGeneration(args.task_id);
   434→  if (result.status === "SUCCESS") {
   435→    return textResult(`Status: SUCCESS\nvideo_url: ${result.videoUrl || "(none)"}\ncover_url: ${result.coverUrl || "(none)"}\n\nDownload the MP4 from video_url.`);
   436→  }
   437→  if (result.status === "FAILED") {
   438→    return textResult(`Status: FAILED\nreason: ${result.error || "unknown"}`, true);
   439→  }
   440→  return textResult(`Status: ${result.status}. Wait ~20-30 seconds and call visioncut_check_video again with the same task_id.`);
   441→}

   454→  mp4: "video/mp4", webm: "video/webm", mov: "video/quicktime", m4v: "video/x-m4v",
   455→  mp3: "audio/mpeg", wav: "audio/wav", m4a: "audio/mp4", aac: "audio/aac", flac: "audio/flac",
   456→  png: "image/png", jpg: "image/jpeg", jpeg: "image/jpeg", webp: "image/webp", gif: "image/gif", avif: "image/avif",
   457→};

   459→function kindOfMime(mimeType: string): "video" | "audio" | "image" | null {
   460→  if (mimeType.startsWith("video/")) return "video";
   461→  if (mimeType.startsWith("audio/")) return "audio";
   462→  if (mimeType.startsWith("image/")) return "image";
   463→  return null;
   464→}

   466→function assertPublicHttpUrl(raw: string): URL {
   467→  let url: URL;
   468→  try {
   469→    url = new URL(raw);
   470→  } catch {
   471→    throw new Error("Invalid URL.");
   472→  }
   473→  if (url.protocol !== "http:" && url.protocol !== "https:") throw new Error("Only http(s) URLs are supported.");
   474→  const h = url.hostname.toLowerCase();
   475→  if (
   476→    h === "localhost" || h.endsWith(".local") || h.endsWith(".internal") || h === "::1" || h === "[::1]" ||
   477→    /^(127\.|10\.|192\.168\.|169\.254\.|0\.)/.test(h) ||
   478→    /^172\.(1[6-9]|2\d|3[01])\./.test(h)
   479→  ) {
   480→    throw new Error("Private/loopback hosts are not allowed.");
   481→  }
   482→  return url;
   483→}

   488→}

   524→async function toolImportMedia(raw: unknown): Promise<McpToolResult> {
   525→  const args = ImportMediaArgs.parse(raw);
   526→  const url = assertPublicHttpUrl(args.url);
   527→  const userId = await demoUserId();
   528→
   529→  const agg = await db.mediaFile.aggregate({ where: { userId }, _sum: { size: true } });
   530→  if ((agg._sum.size ?? 0) >= MAX_USER_CLOUD_TOTAL) {
   531→    return textResult("Import failed: cloud media storage is full (1GB per account). Remove old media first.", true);
   532→  }
   533→
   534→  // Download (size-capped).
   535→  const controller = new AbortController();
   536→  const timer = setTimeout(() => controller.abort(), IMPORT_TIMEOUT_MS);
   537→  let buf: Buffer;
   538→  let mimeType = "";
   539→  try {
   540→    const res = await fetch(url, { signal: controller.signal, redirect: "follow" });
   541→    if (!res.ok) return textResult(`Import failed: the source returned HTTP ${res.status}.`, true);
   542→    const declared = Number(res.headers.get("content-length") || 0);
   543→    if (declared > MAX_IMPORT_FILE) return textResult(`Import failed: file is ${Math.round(declared / 1048576)}MB — the limit is 200MB.`, true);
   544→    mimeType = (res.headers.get("content-type") || "").split(";")[0].trim().toLowerCase();
   545→    buf = Buffer.from(await res.arrayBuffer());
   546→  } catch (err) {
   547→    const aborted = err instanceof Error && err.name === "AbortError";
   548→    return textResult(aborted ? "Import failed: download timed out (60s)." : `Import failed: could not download the file (${err instanceof Error ? err.message : "network error"}).`, true);
   549→  } finally {
   550→    clearTimeout(timer);
   551→  }
   552→  if (!buf.length) return textResult("Import failed: the source returned an empty file.", true);
   553→  if (buf.length > MAX_IMPORT_FILE) return textResult("Import failed: file exceeds the 200MB limit.", true);
   554→
   555→  // Resolve MIME: content-type header, else extension, else reject.
   556→  if (!kindOfMime(mimeType)) {
   557→    const ext = (url.pathname.match(/\.([a-z0-9]+)$/i)?.[1] || "").toLowerCase();
   558→    mimeType = EXT_MIME[ext] || "";
   559→  }
   560→  const kind = kindOfMime(mimeType);
   561→  if (!kind) return textResult(`Import failed: unsupported file type "${mimeType || "unknown"}". Only video, audio and image files are supported.`, true);
   562→
   563→  const name = (args.name || decodeURIComponent(url.pathname.split("/").pop() || "Imported media")).slice(0, 80);
   564→  const assetId = newAssetId();
   565→
   566→  // Quota re-check with the real size, then write MediaFile + 3MB chunks
   567→  // (exactly ceil(size/3MB) chunks — the editor's complete-route contract).
   568→  const reAgg = await db.mediaFile.aggregate({ where: { userId }, _sum: { size: true } });
   569→  if ((reAgg._sum.size ?? 0) + buf.length > MAX_USER_CLOUD_TOTAL) {
   570→    return textResult("Import failed: this file would exceed the 1GB cloud storage quota.", true);
   571→  }
   572→  try {
   573→    await db.mediaFile.create({ data: { id: assetId, userId, name, mimeType, size: buf.length } });
   574→    const chunks: Prisma.MediaChunkCreateManyInput[] = [];
   575→    for (let off = 0, idx = 0; off < buf.length; off += CLOUD_CHUNK, idx++) {
   576→      chunks.push({ fileId: assetId, idx, data: Buffer.from(buf.subarray(off, Math.min(off + CLOUD_CHUNK, buf.length))) });
   577→    }
   578→    await db.mediaChunk.createMany({ data: chunks });
   579→  } catch (err) {
   580→    await db.mediaFile.deleteMany({ where: { id: assetId } }).catch(() => {});
   581→    return textResult(`Import failed while saving: ${err instanceof Error ? err.message.slice(0, 160) : "database error"}.`, true);
   582→  }
   583→
   584→  return textResult(
   585→    `Media imported.\nasset_id: ${assetId}\nname: ${name}\ntype: ${mimeType} (${kind})\nsize: ${buf.length} bytes\n\nPlace it with visioncut_add_to_project (project_id + this asset_id).`,
   586→  );
   587→}

   591→  tracks?: Array<{ id: string; type: "video" | "audio"; name: string; muted: boolean; hidden: boolean; locked: boolean; height: number; clips?: unknown[] }>;
   592→  clips?: Array<Record<string, unknown>>;
   593→  assets?: Array<Record<string, unknown>>;
   594→  duration?: number;
   595→}

   597→async function toolAddToProject(raw: unknown): Promise<McpToolResult> {
   598→  const args = AddToProjectArgs.parse(raw);
   599→  const userId = await demoUserId();
   600→
   601→  const project = await db.project.findFirst({
   602→    where: { id: args.project_id, userId },
   603→    select: { id: true, name: true, data: true },
   604→  });
   605→  if (!project) return textResult(`Project "${args.project_id}" was not found in your account. Use visioncut_list_projects to get valid ids.`, true);
   606→
   607→  const media = await db.mediaFile.findFirst({
   608→    where: { id: args.media_id, userId },
   609→    select: { id: true, name: true, mimeType: true, size: true },
   610→  });
   611→  if (!media) return textResult(`Media "${args.media_id}" was not found in your library. Import it first with visioncut_import_media.`, true);
   612→  const kind = kindOfMime(media.mimeType);
   613→  if (!kind) return textResult(`Media "${media.id}" has an unsupported type (${media.mimeType}).`, true);
   614→
   615→  let state: StoredProjectState;
   616→  try {
   617→    state = JSON.parse(project.data || "{}") as StoredProjectState;
   618→  } catch {
   619→    return textResult("This project's data is corrupted and cannot be updated from here.", true);
   620→  }
   621→  if (!Array.isArray(state.tracks) || !state.tracks.length) {
   622→    state.tracks = [
   623→      { id: "v1", type: "video", name: "Video 1", muted: false, hidden: false, locked: false, height: 64 },
   624→      { id: "a1", type: "audio", name: "Audio 1", muted: false, hidden: false, locked: false, height: 56 },
   625→    ];
   626→  }
   627→  state.assets = Array.isArray(state.assets) ? state.assets : [];
   628→  state.clips = Array.isArray(state.clips) ? state.clips : [];
   629→
   630→  // Register the asset (idempotent by id — the browser restores bytes from /api/media/{id}).
   631→  if (!state.assets.some((a) => a.id === media.id)) {
   632→    state.assets.push({
   633→      id: media.id,
   634→      kind,
   635→      name: media.name,
   636→      mimeType: media.mimeType,
   637→      size: media.size,
   638→      duration: args.duration,
   639→      origin: "ai",
   640→      createdAt: Date.now(),
   641→    });
   642→  }
   643→
   644→  const wantType = args.track_type || (kind === "audio" ? "audio" : "video");
   645→  let track = state.tracks.find((t) => t.type === wantType && !t.locked);
   646→  if (!track) {
   647→    const count = state.tracks.filter((t) => t.type === wantType).length;
   648→    track = {
   649→      id: `${wantType[0]}${Date.now().toString(36).slice(-6)}`,
   650→      type: wantType,
   651→      name: `${wantType === "video" ? "Video" : "Audio"} ${count + 1}`,
   652→      muted: false,
   653→      hidden: false,
   654→      locked: false,
   655→      height: wantType === "video" ? 64 : 56,
   656→    };
   657→    state.tracks.push(track);
   658→  }
   659→
   660→  const trackClips = state.clips.filter((c) => c.trackId === track!.id);
   661→  const trackEnd = trackClips.reduce((m, c) => Math.max(m, Number(c.start || 0) + Number(c.duration || 0)), 0);
   662→  const start = args.at_seconds ?? trackEnd;
   663→  const clipDuration = args.duration ?? (kind === "image" ? 4 : 5);
   664→  const clipId = `c_${Math.random().toString(36).slice(2, 10)}${Date.now().toString(36).slice(-4)}`;
   665→
   666→  state.clips.push({
   667→    id: clipId,
   668→    trackId: track.id,
   669→    type: kind,
   670→    name: media.name,
   671→    assetId: media.id,
   672→    start,
   673→    duration: clipDuration,
   674→    trimStart: 0,
   675→    x: 0,
   676→    y: 0,
   677→    scale: 1,
   678→    rotation: 0,
   679→    opacity: 1,
   680→    volume: 1,
   681→    speed: 1,
   682→    color: { brightness: 1, contrast: 1, saturation: 1, hue: 0, blur: 0 },
   683→    animation: "none",
   684→    animationDuration: 0.6,
   685→  });
   686→  state.duration = state.clips.reduce((m, c) => Math.max(m, Number(c.start || 0) + Number(c.duration || 0)), 0);
   687→
   688→  await db.project.update({ where: { id: project.id }, data: { data: JSON.stringify(state) } });
   689→  return textResult(
   690→    `Clip added to project "${project.name}".\nproject_id: ${project.id}\nclip_id: ${clipId}\ntrack: ${track.name} (${track.type})\nstart: ${start}s, duration: ${clipDuration}s\n\nOpen /editor/${project.id} — the editor restores the media from the cloud automatically.`,
   691→  );
   692→}

   698→export async function callMcpTool(name: string, args: unknown): Promise<McpToolResult> {
   699→  try {
   700→    switch (name) {
   701→      case "visioncut_status":
   702→        StatusArgs.parse(args);
   703→        return await toolStatus();
   704→      case "visioncut_list_projects":
   705→        return await toolListProjects(ListProjectsArgs.parse(args));
   706→      case "visioncut_search_stickers":
   707→        return await toolSearchStickers(args);
   708→      case "visioncut_generate_storyboard":
   709→        return await toolStoryboard(args);
   710→      case "visioncut_generate_image":
   711→        return await toolGenerateImage(args);
   712→      case "visioncut_generate_voiceover":
   713→        return await toolGenerateVoiceover(args);
   714→      case "visioncut_start_video":
   715→        return await toolStartVideo(args);
   716→      case "visioncut_check_video":
   717→        return await toolCheckVideo(args);
   718→      case "visioncut_create_project":
   719→        return await toolCreateProject(args);
   720→      case "visioncut_import_media":
   721→        return await toolImportMedia(args);
   722→      case "visioncut_add_to_project":
   723→        return await toolAddToProject(args);
   724→      default:
   725→        return textResult(`Unknown tool "${name}".`, true);
   726→    }
   727→  } catch (err) {
   728→    if (err instanceof z.ZodError) {
   729→      return textResult(`Invalid arguments for "${name}": ${err.issues.map((i) => `${i.path.join(".") || "args"} — ${i.message}`).join("; ")}`, true);
   730→    }
   731→    return fail(err, name);
   732→  }
   733→}
