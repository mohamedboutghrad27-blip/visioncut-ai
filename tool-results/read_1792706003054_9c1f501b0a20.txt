     1→/**
     2→ * Provider-agnostic text-to-video core for VisionCut AI.
     3→ *
     4→ * Mirrors src/lib/transcribe.ts and src/lib/llm.ts: the video-generation
     5→ * backend is selected via env vars so the same build works in the Z.ai
     6→ * sandbox (platform SDK) and on any hosting with a bring-your-own key
     7→ * (fal.ai, WaveSpeed AI, KIE.AI, PixVerse).
     8→ *
     9→ * Env vars:
    10→ *   VIDEO_PROVIDER         auto (default) | zai | fal | wavespeed | kie | pixverse | higgsfield
    11→ *   VIDEO_API_KEY          Shared fallback key for whichever external provider is selected.
    12→ *   HIGGSFIELD_API_KEY     Per-provider keys — each wins over VIDEO_API_KEY for its own
    13→ *   FAL_API_KEY            platform, so several platforms can coexist and `auto` tries
    14→ *   WAVESPEED_API_KEY      every provider that has a usable key. For higgsfield the key is
    15→ *   KIE_API_KEY            the full "keyId:keySecret" pair from console.higgsfield.ai
    16→ *   PIXVERSE_API_KEY
    17→ *   HIGGSFIELD_VIDEO_MODEL default "alibaba/wan-3.0/text-to-video" (model slug from /models)
    18→ *   FAL_VIDEO_MODEL        default "fal-ai/wan-t2v"  (fal.ai queue model id)
    19→ *   WAVESPEED_VIDEO_MODEL  default "wavespeed-ai/wan-2.1-t2v-480p"
    20→ *   KIE_VIDEO_MODEL        default "veo3-fast"
    21→ *   PIXVERSE_VIDEO_MODEL   default "v4.5"
    22→ *
    23→ * `auto` resolution order:
    24→ *   1. external providers whose credentials are present (VIDEO_API_KEY),
    25→ *      tried in order higgsfield -> fal -> wavespeed -> kie -> pixverse;
    26→ *      auth/billing failures fall through to the next candidate (warnings
    27→ *      are collected on the result so the UI can surface them)
    28→ *   2. `zai` when a Z.ai platform config is available (sandbox)
    29→ *   3. otherwise a clear error listing what was rejected and what to set
    30→ *
    31→ * When a provider is pinned it is used exclusively and its errors surface
    32→ * as-is. Task ids are opaque to the client and are namespaced with the
    33→ * provider ("<provider>:<id>") so /api/ai/video-status can poll the right
    34→ * backend.
    35→ */
    36→

    67→export class VideoError extends Error {
    68→  status: number;
    69→  provider: string;
    70→  constructor(message: string, status = 502, provider = "video") {

    78→const HTTP_TIMEOUT_MS = 30_000;
    79→
    80→function env(name: string): string | undefined {
    81→  const v = process.env[name];
    82→  return v && v.trim() ? v.trim() : undefined;
    83→}
    84→
    85→function videoKey(): string | undefined {
    86→  return env("VIDEO_API_KEY");
    87→}
    88→
    89→/**
    90→ * Per-provider key resolution: a dedicated <PROVIDER>_API_KEY wins, then the
    91→ * shared VIDEO_API_KEY. This lets several platforms coexist — e.g. a Higgsfield
    92→ * key on VIDEO_API_KEY plus a WaveSpeed key on WAVESPEED_API_KEY — and `auto`
    93→ * will try every provider that has a usable key.
    94→ */
    95→function providerKey(provider: VideoProviderName): string | undefined {
    96→  switch (provider) {
    97→    case "higgsfield":
    98→      return env("HIGGSFIELD_API_KEY") || videoKey();
    99→    case "fal":
   100→      return env("FAL_API_KEY") || videoKey();
   101→    case "wavespeed":
   102→      return env("WAVESPEED_API_KEY") || videoKey();
   103→    case "kie":
   104→      return env("KIE_API_KEY") || videoKey();
   105→    case "pixverse":
   106→      return env("PIXVERSE_API_KEY") || videoKey();
   107→    case "zai":
   108→      return undefined; // sandbox SDK manages its own credentials
   109→  }
   110→}
   111→
   112→function providerModel(provider: VideoProviderName): string {
   113→  switch (provider) {
   114→    case "fal":
   115→      return env("FAL_VIDEO_MODEL") || "fal-ai/wan-t2v";
   116→    case "wavespeed":
   117→      return env("WAVESPEED_VIDEO_MODEL") || "wavespeed-ai/wan-2.1-t2v-480p";
   118→    case "kie":
   119→      return env("KIE_VIDEO_MODEL") || "veo3-fast";
   120→    case "pixverse":
   121→      return env("PIXVERSE_VIDEO_MODEL") || "v4.5";
   122→    case "higgsfield":
   123→      return env("HIGGSFIELD_VIDEO_MODEL") || "alibaba/wan-3.0/text-to-video";
   124→    case "zai":
   125→      return "zai-platform-video";
   126→  }
   127→}

   129→/* ------------------------------------------------------------------ */
   130→/* Shared HTTP helper                                                  */
   131→/* ------------------------------------------------------------------ */
   132→
   133→async function httpJson(
   134→  provider: string,
   135→  url: string,
   136→  init: RequestInit & { timeoutMs?: number } = {},
   137→): Promise<{ status: number; body: unknown; text: string }> {
   138→  const controller = new AbortController();
   139→  const timer = setTimeout(() => controller.abort(), init.timeoutMs ?? HTTP_TIMEOUT_MS);
   140→  let res: Response;
   141→  try {
   142→    res = await fetch(url, { ...init, signal: controller.signal });
   143→  } catch (err) {
   144→    const aborted = err instanceof Error && err.name === "AbortError";
   145→    throw new VideoError(
   146→      aborted
   147→        ? `${provider} request timed out.`
   148→        : `Could not reach ${provider} (${new URL(url).host}). Check network.`,
   149→      aborted ? 504 : 502,
   150→      provider,
   151→    );
   152→  } finally {
   153→    clearTimeout(timer);
   154→  }
   155→  const text = await res.text();
   156→  let body: unknown = null;
   157→  try {
   158→    body = JSON.parse(text);
   159→  } catch {
   160→    /* keep text */
   161→  }
   162→  if (!res.ok) {
   163→    const snippet = text.replace(/\s+/g, " ").slice(0, 220);
   164→    const map: Record<number, [number, string]> = {
   165→      401: [502, `The ${provider} API key was rejected (401). Check VIDEO_API_KEY.`],
   166→      402: [402, `${provider} account has insufficient credits (402).`],
   167→      403: [502, `The ${provider} key is not allowed for this request (403).`],
   168→      404: [502, `${provider} model or endpoint not found (404). Check the video model id.`],
   169→      429: [429, `${provider} rate limit exceeded (429).`],
   170→    };
   171→    const mapped = map[res.status];
   172→    if (mapped) throw new VideoError(`${mapped[1]} ${snippet}`.trim(), mapped[0], provider);
   173→    throw new VideoError(
   174→      `${provider} request failed (HTTP ${res.status}). ${snippet}`.trim(),
   175→      502,
   176→      provider,
   177→    );
   178→  }
   179→  return { status: res.status, body, text };
   180→}

   186→interface ZaiTaskLike {
   187→  id?: string;
   188→  task_status?: string;

   190→
   191→async function submitZai(opts: VideoSubmitOptions): Promise<VideoSubmitResult> {
   192→  let zai: Awaited<ReturnType<typeof ZAI.create>>;
   193→  try {
   194→    zai = await ZAI.create();
   195→  } catch {
   196→    throw new VideoError(
   197→      "Video generation has no provider on this server (the built-in engine only works in its own sandbox). " +
   198→        "Set VIDEO_PROVIDER to your video API platform (fal | wavespeed | kie | pixverse) with a valid VIDEO_API_KEY.",
   199→      503,
   200→      "zai",
   201→    );
   202→  }
   203→  const task = (await zai.video.generations.create({
   204→    prompt: opts.prompt,
   205→    quality: opts.quality,
   206→    with_audio: opts.withAudio,
   207→    fps: opts.fps,
   208→    duration: opts.duration,
   209→  })) as ZaiTaskLike;
   210→  if (!task?.id) {
   211→    throw new VideoError("The video service did not accept the task. Please try again.", 502, "zai");
   212→  }
   213→  return { taskId: `zai:${task.id}`, provider: "zai", model: "zai-platform-video" };
   214→}
   215→
   216→interface ZaiResultLike {
   217→  task_status?: string;
   218→  video_result?: Array<{ url?: string; cover_image_url?: string }>;
   219→  video_url?: string;
   220→  url?: string;
   221→  video?: string;
   222→}
   223→
   224→async function pollZai(id: string): Promise<VideoPollResult> {
   225→  const zai = await ZAI.create();
   226→  const result = (await zai.async.result.query(id)) as ZaiResultLike;
   227→  if (result.task_status === "SUCCESS") {
   228→    const url =
   229→      result.video_result?.[0]?.url || result.video_url || result.url || result.video || null;
   230→    if (!url) {
   231→      return { status: "FAILED", error: "Generation finished but no video was returned." };
   232→    }
   233→    return {
   234→      status: "SUCCESS",
   235→      videoUrl: url,
   236→      coverUrl: result.video_result?.[0]?.cover_image_url || undefined,
   237→    };
   238→  }
   239→  if (result.task_status === "FAIL" || result.task_status === "FAILED") {
   240→    return { status: "FAILED", error: "The video generation task failed." };
   241→  }
   242→  return { status: "PROCESSING" };
   243→}

   246→/* fal.ai (queue API — https://docs.fal.ai)                            */
   247→/* ------------------------------------------------------------------ */
   248→
   249→async function submitFal(opts: VideoSubmitOptions, key: string, model: string): Promise<VideoSubmitResult> {
   250→  const numFrames = opts.duration === 10 ? 161 : 81;
   251→  const { body } = await httpJson("fal", `https://queue.fal.run/${model}`, {
   252→    method: "POST",
   253→    headers: {
   254→      Authorization: `Key ${key}`,
   255→      "Content-Type": "application/json",
   256→    },
   257→    body: JSON.stringify({
   258→      prompt: opts.prompt,
   259→      aspect_ratio: "16:9",
   260→      resolution: "480p",
   261→      num_frames: numFrames,
   262→      enable_safety_checker: true,
   263→    }),
   264→  });
   265→  const requestId = (body as { request_id?: string } | null)?.request_id;
   266→  if (typeof requestId !== "string" || !requestId) {
   267→    throw new VideoError("fal did not return a request id.", 502, "fal");
   268→  }
   269→  return { taskId: `fal:${requestId}`, provider: "fal", model };
   270→}
   271→
   272→interface FalStatusBody {
   273→  status?: string;
   274→  queue_position?: number;
   275→  error?: unknown;
   276→}
   277→
   278→async function pollFal(id: string, key: string): Promise<VideoPollResult> {
   279→  const headers = { Authorization: `Key ${key}` };
   280→  const { body: statusBody } = await httpJson(
   281→    "fal",
   282→    `https://queue.fal.run/requests/${encodeURIComponent(id)}/status?logs=0`,
   283→    { headers },
   284→  );
   285→  const st = (statusBody as FalStatusBody | null)?.status;
   286→  if (st === "COMPLETED") {
   287→    const { body } = await httpJson(
   288→      "fal",
   289→      `https://queue.fal.run/requests/${encodeURIComponent(id)}`,
   290→      { headers },
   291→    );
   292→    const b = body as { video?: { url?: string }; videos?: Array<{ url?: string }>; url?: string } | null;
   293→    const url = b?.video?.url || b?.videos?.[0]?.url || b?.url || null;
   294→    if (!url) return { status: "FAILED", error: "fal finished but no video URL was returned." };
   295→    return { status: "SUCCESS", videoUrl: url };
   296→  }
   297→  if (st === "IN_QUEUE") return { status: "PENDING" };
   298→  if (st === "IN_PROGRESS") return { status: "PROCESSING" };
   299→  return { status: "PROCESSING" };
   300→}

   306→async function submitWavespeed(
   307→  opts: VideoSubmitOptions,
   308→  key: string,
   309→  model: string,
   310→): Promise<VideoSubmitResult> {
   311→  const { body } = await httpJson("wavespeed", `https://api.wavespeed.ai/api/v3/${model}`, {
   312→    method: "POST",
   313→    headers: { Authorization: `Bearer ${key}`, "Content-Type": "application/json" },
   314→    body: JSON.stringify({ prompt: opts.prompt, aspect_ratio: "16:9" }),
   315→  });
   316→  const data = (body as { data?: { id?: string } } | null)?.data;
   317→  if (!data?.id) throw new VideoError("WaveSpeed did not return a task id.", 502, "wavespeed");
   318→  return { taskId: `wavespeed:${data.id}`, provider: "wavespeed", model };
   319→}
   320→
   321→interface WaveSpeedResultBody {
   322→  data?: { status?: string; outputs?: Array<string | { url?: string }>; error?: string };
   323→}
   324→
   325→async function pollWavespeed(id: string, key: string): Promise<VideoPollResult> {
   326→  const { body } = await httpJson(
   327→    "wavespeed",
   328→    `https://api.wavespeed.ai/api/v3/predictions/${encodeURIComponent(id)}/result`,
   329→    { headers: { Authorization: `Bearer ${key}` } },
   330→  );
   331→  const data = (body as WaveSpeedResultBody | null)?.data;
   332→  const status = data?.status;
   333→  if (status === "completed") {
   334→    const outputs = data?.outputs || [];
   335→    const first = outputs[0];
   336→    const url = typeof first === "string" ? first : first?.url;
   337→    if (!url) return { status: "FAILED", error: "WaveSpeed completed but returned no video." };
   338→    return { status: "SUCCESS", videoUrl: url };
   339→  }
   340→  if (status && ["failed", "cancelled", "timeout", "deleted"].includes(status)) {
   341→    return { status: "FAILED", error: data?.error || `WaveSpeed task ${status}.` };
   342→  }
   343→  if (status === "created") return { status: "PENDING" };
   344→  return { status: "PROCESSING" };
   345→}

   348→/* KIE.AI (jobs API — https://docs.kie.ai)                             */
   349→/* ------------------------------------------------------------------ */
   350→
   351→async function submitKie(opts: VideoSubmitOptions, key: string, model: string): Promise<VideoSubmitResult> {
   352→  const { body } = await httpJson("kie", "https://api.kie.ai/api/v1/jobs/createTask", {
   353→    method: "POST",
   354→    headers: { Authorization: `Bearer ${key}`, "Content-Type": "application/json" },
   355→    body: JSON.stringify({ model, input: { prompt: opts.prompt } }),
   356→  });
   357→  const body2 = body as { code?: number; msg?: string; data?: { taskId?: string } } | null;
   358→  if (body2?.code && body2.code !== 200) {
   359→    if (body2.code === 401) {
   360→      throw new VideoError("The kie API key was rejected (401). Check VIDEO_API_KEY.", 502, "kie");
   361→    }
   362→    throw new VideoError(`KIE.AI error ${body2.code}: ${body2.msg || "unknown"}`, 502, "kie");
   363→  }
   364→  const data = body2?.data;
   365→  if (!data?.taskId) throw new VideoError("KIE.AI did not return a task id.", 502, "kie");
   366→  return { taskId: `kie:${data.taskId}`, provider: "kie", model };
   367→}
   368→
   369→interface KieResultBody {
   370→  data?: { state?: string; resultJson?: string; failMsg?: string };
   371→}
   372→
   373→async function pollKie(id: string, key: string): Promise<VideoPollResult> {
   374→  const { body } = await httpJson(
   375→    "kie",
   376→    `https://api.kie.ai/api/v1/jobs/recordInfo?taskId=${encodeURIComponent(id)}`,
   377→    { headers: { Authorization: `Bearer ${key}` } },
   378→  );
   379→  const data = (body as KieResultBody | null)?.data;
   380→  const state = data?.state;
   381→  if (state === "success") {
   382→    let url: string | null = null;
   383→    try {
   384→      const parsed = JSON.parse(data?.resultJson || "{}") as { resultUrls?: string[] };
   385→      url = parsed.resultUrls?.[0] || null;
   386→    } catch {
   387→      /* ignore */
   388→    }
   389→    if (!url) return { status: "FAILED", error: "KIE.AI finished but returned no video URL." };
   390→    return { status: "SUCCESS", videoUrl: url };
   391→  }
   392→  if (state === "fail") return { status: "FAILED", error: data?.failMsg || "KIE.AI task failed." };
   393→  if (state === "waiting" || state === "queuing") return { status: "PENDING" };
   394→  return { status: "PROCESSING" };
   395→}

   398→/* PixVerse (openapi v2 — https://docs.platform.pixverse.ai)           */
   399→/* ------------------------------------------------------------------ */
   400→
   401→async function submitPixverse(
   402→  opts: VideoSubmitOptions,
   403→  key: string,
   404→  model: string,
   405→): Promise<VideoSubmitResult> {
   406→  const { body } = await httpJson(
   407→    "pixverse",
   408→    "https://app-api.pixverse.ai/openapi/v2/video/text/generate",
   409→    {
   410→      method: "POST",
   411→      headers: {
   412→        "API-KEY": key,
   413→        "Ai-Trace-Id": `visioncut-${Date.now()}`,
   414→        "Content-Type": "application/json",
   415→      },
   416→      body: JSON.stringify({
   417→        prompt: opts.prompt,
   418→        aspect_ratio: "16:9",
   419→        duration: String(opts.duration),
   420→        quality: "540p",
   421→        ai_model: model,
   422→      }),
   423→    },
   424→  );
   425→  const root = body as { ErrCode?: number; ErrMsg?: string; Resp?: { video_id?: number | string } } | null;
   426→  if (root?.ErrCode && root.ErrCode !== 0) {
   427→    if (root.ErrCode === 10005) {
   428→      throw new VideoError("The pixverse API key is not registered (rejected). Check VIDEO_API_KEY.", 502, "pixverse");
   429→    }
   430→    throw new VideoError(`PixVerse error ${root.ErrCode}: ${root.ErrMsg || "unknown"}`, 502, "pixverse");
   431→  }
   432→  const resp = root?.Resp;
   433→  if (!resp?.video_id) throw new VideoError("PixVerse did not return a video id.", 502, "pixverse");
   434→  return { taskId: `pixverse:${resp.video_id}`, provider: "pixverse", model };
   435→}
   436→
   437→interface PixverseResultBody {
   438→  ErrCode?: number;
   439→  ErrMsg?: string;
   440→  Resp?: { status?: number; url?: string; cover_url?: string; msg?: string };
   441→}
   442→
   443→async function pollPixverse(id: string, key: string): Promise<VideoPollResult> {
   444→  const { body } = await httpJson(
   445→    "pixverse",
   446→    `https://app-api.pixverse.ai/openapi/v2/video/result/${encodeURIComponent(id)}`,
   447→    { headers: { "API-KEY": key, "Ai-Trace-Id": `visioncut-${Date.now()}` } },
   448→  );
   449→  const root = body as PixverseResultBody | null;
   450→  if (root?.ErrCode && root.ErrCode !== 0) {
   451→    return { status: "FAILED", error: root.ErrMsg || `PixVerse error ${root.ErrCode}.` };
   452→  }
   453→  const resp = root?.Resp;
   454→  // PixVerse status codes: 1 = generating, 5 = success, 7 = content-moderation failure
   455→  if (resp?.status === 5) {
   456→    if (!resp.url) return { status: "FAILED", error: "PixVerse finished but returned no video URL." };
   457→    return { status: "SUCCESS", videoUrl: resp.url, coverUrl: resp.cover_url || undefined };
   458→  }
   459→  if (resp?.status === 7) {
   460→    return { status: "FAILED", error: resp.msg || "PixVerse rejected the prompt (moderation)." };
   461→  }
   462→  return { status: "PROCESSING" };
   463→}

   466→/* Higgsfield (queue API — https://docs.higgsfield.ai)                  */
   467→/* VIDEO_API_KEY = "keyId:keySecret", sent as `Authorization: Key ...`   */

   469→
   470→async function submitHiggsfield(
   471→  opts: VideoSubmitOptions,
   472→  key: string,
   473→  model: string,
   474→): Promise<VideoSubmitResult> {
   475→  let body: unknown;
   476→  try {
   477→    ({ body } = await httpJson("higgsfield", `https://api.higgsfield.ai/${model}`, {
   478→      method: "POST",
   479→      headers: {
   480→        Authorization: `Key ${key}`,
   481→        "Content-Type": "application/json",
   482→        "Idempotency-Key": crypto.randomUUID(),
   483→      },
   484→      body: JSON.stringify({
   485→        prompt: opts.prompt,
   486→        duration: opts.duration,
   487→        resolution: opts.quality === "quality" ? "1080p" : "720p",
   488→        aspect_ratio: "16:9",
   489→        generate_audio: opts.withAudio,
   490→      }),
   491→    }));
   492→  } catch (err) {
   493→    if (err instanceof VideoError && /not_enough_credits/i.test(err.message)) {
   494→      throw new VideoError(
   495→        "Higgsfield account is out of credits — top up at console.higgsfield.ai, then try again.",
   496→        402,
   497→        "higgsfield",
   498→      );
   499→    }
   500→    throw err;
   501→  }
   502→  const requestId = (body as { request_id?: string } | null)?.request_id;
   503→  if (typeof requestId !== "string" || !requestId) {
   504→    throw new VideoError("Higgsfield did not return a request id.", 502, "higgsfield");
   505→  }
   506→  return { taskId: `higgsfield:${requestId}`, provider: "higgsfield", model };
   507→}

   509→interface HiggsfieldStatusBody {
   510→  status?: "queued" | "in_progress" | "nsfw" | "failed" | "completed" | "canceled";
   511→  error?: string | null;
   512→  video?: { url?: string } | null;
   513→  images?: Array<{ url?: string }> | null;
   514→  audios?: Array<{ url?: string }> | null;
   515→}
   516→
   517→async function pollHiggsfield(id: string, key: string): Promise<VideoPollResult> {
   518→  const { body } = await httpJson(
   519→    "higgsfield",
   520→    `https://api.higgsfield.ai/requests/${encodeURIComponent(id)}/status`,
   521→    { headers: { Authorization: `Key ${key}` } },
   522→  );
   523→  const b = (body as HiggsfieldStatusBody | null) || {};
   524→  switch (b.status) {
   525→    case "completed": {
   526→      const url = b.video?.url || b.images?.[0]?.url || b.audios?.[0]?.url || null;
   527→      if (!url) return { status: "FAILED", error: "Higgsfield finished but returned no video URL." };
   528→      return { status: "SUCCESS", videoUrl: url };
   529→    }
   530→    case "failed":
   531→      return { status: "FAILED", error: b.error || "Higgsfield task failed." };
   532→    case "nsfw":
   533→      return { status: "FAILED", error: "Higgsfield rejected the prompt (content moderation)." };
   534→    case "canceled":
   535→      return { status: "FAILED", error: "Higgsfield task was canceled." };
   536→    case "queued":
   537→      return { status: "PENDING" };
   538→    default:
   539→      return { status: "PROCESSING" };
   540→  }
   541→}

   543→/* ------------------------------------------------------------------ */
   544→/* Resolution chain                                                    */
   545→/* ------------------------------------------------------------------ */
   546→
   547→function providerOrder(selected: VideoProviderSetting): VideoProviderName[] {
   548→  if (selected !== "auto") return [selected];
   549→  const order: VideoProviderName[] = [];
   550→  for (const p of ["higgsfield", "fal", "wavespeed", "kie", "pixverse"] as VideoProviderName[]) {
   551→    if (providerKey(p)) order.push(p);
   552→  }
   553→  order.push("zai");
   554→  return order;
   555→}
   556→
   557→async function submitProvider(
   558→  provider: VideoProviderName,
   559→  opts: VideoSubmitOptions,
   560→): Promise<VideoSubmitResult> {
   561→  const key = providerKey(provider);
   562→  const model = providerModel(provider);
   563→  switch (provider) {
   564→    case "fal":
   565→      if (!key) throw new VideoError("VIDEO_API_KEY is not configured for fal.", 503, "fal");
   566→      return submitFal(opts, key, model);
   567→    case "wavespeed":
   568→      if (!key) throw new VideoError("VIDEO_API_KEY is not configured for wavespeed.", 503, "wavespeed");
   569→      return submitWavespeed(opts, key, model);
   570→    case "kie":
   571→      if (!key) throw new VideoError("VIDEO_API_KEY is not configured for kie.", 503, "kie");
   572→      return submitKie(opts, key, model);
   573→    case "pixverse":
   574→      if (!key) throw new VideoError("VIDEO_API_KEY is not configured for pixverse.", 503, "pixverse");
   575→      return submitPixverse(opts, key, model);
   576→    case "higgsfield":
   577→      if (!key) throw new VideoError("VIDEO_API_KEY is not configured for higgsfield.", 503, "higgsfield");
   578→      return submitHiggsfield(opts, key, model);
   579→    case "zai":
   580→      return submitZai(opts);
   581→  }
   582→}

   584→/**
   585→ * Submit a text-to-video task using the configured provider chain.
   586→ * In `auto` mode, auth/billing failures on external providers fall through
   587→ * to the next candidate; pinned providers surface errors as-is.
   588→ */
   589→export async function submitVideoGeneration(
   590→  opts: VideoSubmitOptions,
   591→): Promise<VideoSubmitResult> {
   592→  const selected = (env("VIDEO_PROVIDER") as VideoProviderSetting | undefined) || "auto";
   593→  const order = providerOrder(selected);
   594→  const pinned = selected !== "auto";
   595→  const failures: string[] = [];
   596→  const warnings: string[] = [];
   597→
   598→  for (const provider of order) {
   599→    try {
   600→      const result = await submitProvider(provider, opts);
   601→      if (warnings.length) result.warnings = warnings;
   602→      return result;
   603→    } catch (err) {
   604→      const ve = err instanceof VideoError ? err : new VideoError(String(err), 502, provider);
   605→      // Sandbox-only provider (zai) outside the sandbox or any pinned failure stops the chain.
   606→      if (pinned) throw ve;
   607→      const short = ve.message.replace(/\s+/g, " ").slice(0, 180);
   608→      failures.push(`${provider}: ${ve.message}`);
   609→      warnings.push(`${provider}: ${short}`);
   610→      console.warn(`[video] provider ${provider} failed (status ${ve.status}), trying next...`);
   611→    }
   612→  }
   613→  throw new VideoError(
   614→    "No video provider accepted the task. " +
   615→      (failures.length
   616→        ? `Provider results — ${failures.join(" | ")}`
   617→        : "Set VIDEO_PROVIDER and VIDEO_API_KEY, or use the Z.ai sandbox."),
   618→    503,
   619→    "video",
   620→  );
   621→}

   623→/** Poll a previously submitted task ("<provider>:<id>"). */
   624→export async function pollVideoGeneration(taskId: string): Promise<VideoPollResult> {
   625→  const sep = taskId.indexOf(":");
   626→  const provider = (sep === -1 ? "zai" : taskId.slice(0, sep)) as VideoProviderName;
   627→  const id = sep === -1 ? taskId : taskId.slice(sep + 1);
   628→  const key = providerKey(provider);
   629→
   630→  if (!/^[a-z0-9A-Z_\-\.]+$/.test(id)) {
   631→    throw new VideoError("Malformed task id.", 400, "video");
   632→  }
   633→
   634→  switch (provider) {
   635→    case "zai":
   636→      return pollZai(id);
   637→    case "fal":
   638→      if (!key) throw new VideoError("VIDEO_API_KEY is not configured for fal.", 503, "fal");
   639→      return pollFal(id, key);
   640→    case "wavespeed":
   641→      if (!key) throw new VideoError("VIDEO_API_KEY is not configured for wavespeed.", 503, "wavespeed");
   642→      return pollWavespeed(id, key);
   643→    case "kie":
   644→      if (!key) throw new VideoError("VIDEO_API_KEY is not configured for kie.", 503, "kie");
   645→      return pollKie(id, key);
   646→    case "pixverse":
   647→      if (!key) throw new VideoError("VIDEO_API_KEY is not configured for pixverse.", 503, "pixverse");
   648→      return pollPixverse(id, key);
   649→    case "higgsfield":
   650→      if (!key) throw new VideoError("VIDEO_API_KEY is not configured for higgsfield.", 503, "higgsfield");
   651→      return pollHiggsfield(id, key);
   652→    default:
   653→      throw new VideoError(`Unknown video provider "${provider}" in task id.`, 400, "video");
   654→  }
   655→}
