   139→   139→        },
   140→   140→        { select: false }

   143→   143→    }

   173→   173→      }
   174→   174→      toast.success(`Scene ${scene.index} visual added to the timeline`);
   175→   175→    } catch (err) {
   176→   176→      toast.error(err instanceof Error ? err.message : "Scene visual failed");
   177→   177→    } finally {
   178→   178→      setGenImage(null);
   179→   179→    }
   180→   180→  };
   181→   181→
   182→   182→  return (
   183→   183→    <div>
   184→   184→      <textarea
   185→   185→        value={script}
   186→   186→        onChange={(e) => setScript(e.target.value)}
   187→   187→        rows={5}
   188→   188→        maxLength={6000}

   190→   190→        className="scrollbar-slim w-full resize-none rounded-xl border border-white/10 bg-black/40 p-3 text-[11px] leading-relaxed text-zinc-200 placeholder:text-zinc-600 focus:border-fuchsia-400/50 focus:outline-none"

   201→   201→          ))}
   202→   202→        </select>
   203→   203→        <button
   204→   204→          onClick={run}
   205→   205→          disabled={busy}
   206→   206→          className="inline-flex h-8 shrink-0 items-center gap-1.5 rounded-lg bg-gradient-to-r from-fuchsia-600 to-violet-500 px-3 text-[11px] font-semibold text-white transition-all hover:brightness-110 disabled:opacity-60"

   260→   260→function TextToVideoTool() {
   261→   261→  const [prompt, setPrompt] = useState("");
   262→   262→  const [duration, setDuration] = useState<5 | 10>(5);
   263→   263→  const [quality, setQuality] = useState<"speed" | "quality">("speed");
   264→   264→  const [withAudio, setWithAudio] = useState(false);
   265→   265→  const [busy, setBusy] = useState(false);
   266→   266→  const [status, setStatus] = useState<string | null>(null);
   267→   267→  const [pollCount, setPollCount] = useState(0);
   268→   268→
   269→   269→  const run = async () => {
   270→   270→    if (prompt.trim().length < 3) {
   271→   271→      toast.error("Describe the video you want first.");
   272→   272→      return;
   273→   273→    }
   274→   274→    setBusy(true);
   275→   275→    setPollCount(0);
   276→   276→    setStatus("Submitting generation task…");
   277→   277→    try {
   278→   278→      const res = await fetch("/api/ai/video", {
   279→   279→        method: "POST",
   280→   280→        headers: { "Content-Type": "application/json" },
   281→   281→        body: JSON.stringify({ prompt: prompt.trim(), duration, quality, withAudio, fps: 30 }),
   282→   282→      });
   283→   283→      const data = await res.json();
   284→   284→      if (!res.ok) throw new Error(data.error || "Could not start video generation");
   285→   285→      const taskId = data.taskId as string;
   286→   286→
   287→   287→      // Poll until done (max ~7 min)
   288→   288→      const maxPolls = 84;
   289→   289→      let videoUrl: string | null = null;
   290→   290→      for (let i = 1; i <= maxPolls; i++) {
   291→   291→        setPollCount(i);
   292→   292→        setStatus(`Generating… check ${i}/${maxPolls} (typically 1–4 min)`);
   293→   293→        await new Promise((r) => setTimeout(r, 5000));
   294→   294→        const sres = await fetch(`/api/ai/video-status/${taskId}`);
   295→   295→        const sdata = await sres.json();
   296→   296→        if (sdata.status === "SUCCESS") {
   297→   297→          videoUrl = sdata.videoUrl;
   298→   298→          break;
   299→   299→        }
   300→   300→        if (sdata.status === "FAILED") throw new Error(sdata.error || "Generation failed on the provider side.");
   301→   301→      }
   302→   302→      if (!videoUrl) throw new Error("Generation is taking longer than expected — try again later.");
   303→   303→
   304→   304→      setStatus("Downloading generated video…");
   305→   305→      const proxied = await fetch(`/api/ai/proxy-media?url=${encodeURIComponent(videoUrl)}`);
   306→   306→      if (!proxied.ok) throw new Error("Could not download the generated video.");
   307→   307→      const blob = await proxied.blob();
   308→   308→
   309→   309→      // probe duration
   310→   310→      const probeUrl = URL.createObjectURL(blob);
   311→   311→      const dur = await new Promise<number>((resolve) => {
   312→   312→        const v = document.createElement("video");
   313→   313→        v.preload = "metadata";
   314→   314→        v.src = probeUrl;
   315→   315→        v.onloadedmetadata = () => resolve(isFinite(v.duration) ? v.duration : 5);
   316→   316→        v.onerror = () => resolve(5);
   317→   317→        setTimeout(() => resolve(5), 8000);
   318→   318→      });
   319→   319→
   320→   320→      setStatus("Adding to timeline…");
   321→   321→      const { asset } = await ingestBlob(blob, {
   322→   322→        name: prompt.trim().slice(0, 32) + "…",
   323→   323→        kind: "video",
   324→   324→        mimeType: blob.type || "video/mp4",
   325→   325→        duration: dur,
   326→   326→        origin: "ai",
   327→   327→      });
   328→   328→      await new Promise<void>((resolve) => {
   329→   329→        const v = document.createElement("video");
   330→   330→        v.src = probeUrl;
   331→   331→        v.muted = true;
   332→   332→        v.currentTime = Math.min(1, dur * 0.1);
   333→   333→        v.onseeked = () => {
   334→   334→          const c = document.createElement("canvas");
   335→   335→          const scale = Math.min(1, 320 / (v.videoWidth || 320));
   336→   336→          c.width = Math.round((v.videoWidth || 320) * scale);
   337→   337→          c.height = Math.round((v.videoHeight || 180) * scale);
   338→   338→          c.getContext("2d")?.drawImage(v, 0, 0, c.width, c.height);
   339→   339→          const store = useEditor.getState();
   340→   340→          const assets = store.project.assets.map((a) => (a.id === asset.id ? { ...a, thumbnail: c.toDataURL("image/jpeg", 0.6) } : a));
   341→   341→          useEditor.setState({ project: { ...store.project, assets } });
   342→   342→          resolve();
   343→   343→        };
   344→   344→        setTimeout(resolve, 8000);
   345→   345→      });
   346→   346→      URL.revokeObjectURL(probeUrl);
   347→   347→
   348→   348→      const store = useEditor.getState();
   349→   349→      store.addAsset(asset); // register asset metadata in the project
   350→   350→      const url = await getAssetUrl(asset.id);
   351→   351→      const clip = store.addClipFromAsset(asset);
   352→   352→      if (clip && url) ensureMediaUrl(clip, url);
   353→   353→      toast.success("AI video added to the timeline");
   354→   354→      setStatus(null);
   355→   355→    } catch (err) {
   356→   356→      toast.error(err instanceof Error ? err.message : "Video generation failed");
   357→   357→      setStatus(null);
   358→   358→    } finally {
   359→   359→      setBusy(false);
   360→   360→    }
   361→   361→  };
   362→   362→
   363→   363→  return (
   364→   364→    <div>
   365→   365→      <textarea
   366→   366→        value={prompt}
   367→   367→        onChange={(e) => setPrompt(e.target.value)}
   368→   368→        rows={4}
   369→   369→        maxLength={1500}
   370→   370→        placeholder="e.g. A neon-lit cyberpunk street in the rain, camera slowly dollying forward, reflections on wet asphalt"
   371→   371→        className="scrollbar-slim w-full resize-none rounded-xl border border-white/10 bg-black/40 p-3 text-[11px] leading-relaxed text-zinc-200 placeholder:text-zinc-600 focus:border-fuchsia-400/50 focus:outline-none"
   372→   372→      />
   373→   373→      <div className="mt-2 grid grid-cols-3 gap-1.5">
   374→   374→        <label className="rounded-lg border border-white/10 bg-black/30 px-2 py-1.5 text-[10px] text-zinc-400">
   375→   375→          Duration
   376→   376→          <select
   377→   377→            value={duration}
   378→   378→            onChange={(e) => setDuration(Number(e.target.value) as 5 | 10)}
   379→   379→            className="mt-0.5 block w-full bg-transparent text-[11px] font-semibold text-zinc-200 focus:outline-none"
   380→   380→          >
   381→   381→            <option value={5} className="bg-[#101018]">5s</option>
   382→   382→            <option value={10} className="bg-[#101018]">10s</option>
   383→   383→          </select>
   384→   384→        </label>
   385→   385→        <label className="rounded-lg border border-white/10 bg-black/30 px-2 py-1.5 text-[10px] text-zinc-400">
   386→   386→          Quality
   387→   387→          <select
   388→   388→            value={quality}
   389→   389→            onChange={(e) => setQuality(e.target.value as "speed" | "quality")}
   390→   390→            className="mt-0.5 block w-full bg-transparent text-[11px] font-semibold text-zinc-200 focus:outline-none"
   391→   391→          >
   392→   392→            <option value="speed" className="bg-[#101018]">Fast</option>
   393→   393→            <option value="quality" className="bg-[#101018]">Detailed</option>
   394→   394→          </select>
   395→   395→        </label>
   396→   396→        <button
   397→   397→          onClick={() => setWithAudio((v) => !v)}
   398→   398→          className={cn(
   399→   399→            "rounded-lg border px-2 py-1.5 text-left text-[10px] transition-colors",

   696→   696→/* ------------------------------------------------------------------ */
   697→   697→/* Smart editing                                                        */
   698→   698→/* ------------------------------------------------------------------ */
   699→   699→
   700→   700→function SmartTools() {
   701→   701→  const selection = useEditor((s) => s.selection);
   702→   702→  const clip = useEditor((s) => s.project.clips.find((c) => c.id === s.selection));
   703→   703→  const [busy, setBusy] = useState(false);
   704→   704→
   705→   705→  const removeSilences = async () => {
   706→   706→    if (!clip || (clip.type !== "video" && clip.type !== "audio")) {
   707→   707→      toast.error("Select a video or audio clip with audio first.");
   708→   708→      return;
   709→   709→    }
   710→   710→    const peaks = clip.peaks;
   711→   711→    if (!peaks || peaks.length === 0) {
   712→   712→      toast.error("No waveform data yet — wait a moment for analysis, then try again.");
   713→   713→      return;
   714→   714→    }
   715→   715→    setBusy(true);
   716→   716→    try {
   717→   717→      const store = useEditor.getState();
   718→   718→      const asset = store.project.assets.find((a) => a.id === clip.assetId);
   719→   719→      const sourceDur = asset?.duration ?? clip.duration;
   720→   720→      const bucketDur = sourceDur / peaks.length;
   721→   721→      const THRESHOLD = 0.055;
   722→   722→      const MIN_SILENCE = Math.max(1, Math.round(0.45 / bucketDur));
   723→   723→      const MIN_KEEP = Math.max(1, Math.round(0.3 / bucketDur));
   724→   724→
   725→   725→      // find silent runs
   726→   726→      const silent: boolean[] = peaks.map((p) => p < THRESHOLD);
   727→   727→      const keepSegments: { start: number; end: number }[] = [];
   728→   728→      let i = 0;
   729→   729→      let cursor = 0; // source seconds
   730→   730→      while (i < silent.length) {
   731→   731→        if (silent[i]) {
   732→   732→          let j = i;
   733→   733→          while (j < silent.length && silent[j]) j++;
   734→   734→          const silenceLen = j - i;
   735→   735→          if (silenceLen >= MIN_SILENCE) {
   736→   736→            const silenceStart = i * bucketDur;
   737→   737→            const silenceEnd = j * bucketDur;
   738→   738→            if (silenceStart - cursor > 0.05) {
   739→   739→              keepSegments.push({ start: cursor, end: silenceStart });
   740→   740→            }
   741→   741→            cursor = silenceEnd;
   742→   742→          }
   743→   743→          i = j;
   744→   744→        } else i++;
   745→   745→      }
   746→   746→      if (cursor < sourceDur - 0.05) keepSegments.push({ start: cursor, end: sourceDur });
   747→   747→      const meaningful = keepSegments.filter((s) => s.end - s.start >= 0.15);
   748→   748→      if (meaningful.length === 0) {
   749→   749→        toast.info("No significant silence found in this clip.");
   750→   750→        return;
   751→   751→      }
   752→   752→
   753→   753→      store.commit();
   754→   754→      // replace the original clip with keep-segments
   755→   755→      const trackId = clip.trackId;
   756→   756→      let timelineT = clip.start;
   757→   757→      const newClips: Clip[] = meaningful.map((seg, idx) => ({
   758→   758→        ...structuredClone(clip),
   759→   759→        id: uid(),
   760→   760→        name: idx === 0 ? clip.name : `${clip.name} (${idx + 1})`,
   761→   761→        start: timelineT,
   762→   762→        duration: (seg.end - seg.start) / clip.speed,
   763→   763→        trimStart: clip.trimStart + seg.start,
   764→   764→        peaks: undefined,
   765→   765→      }));
   766→   766→      for (const c of newClips) timelineT += c.duration;
   767→   767→
   768→   768→      const rest = store.project.clips.filter((c) => c.id !== clip.id);
   769→   769→      // shift subsequent clips on the same track by the removed time
   770→   770→      const removedTotal = clip.duration - (timelineT - clip.start);
   771→   771→      const shifted = rest.map((c) =>
   772→   772→        c.trackId === trackId && c.start >= clip.start + clip.duration - 1e-3
   773→   773→          ? { ...c, start: Math.max(0, c.start - removedTotal) }
   774→   774→          : c
   775→   775→      );
   776→   776→      useEditor.setState({
   777→   777→        project: {
   778→   778→          ...store.project,
   779→   779→          clips: [...shifted, ...newClips],
   780→   780→          duration: Math.max(0, store.project.duration - removedTotal),
   781→   781→        },
   782→   782→        saveStatus: "unsaved",
   783→   783→      });
   784→   784→      toast.success(`Removed ${newClips.length - 1 || "no"} silence gap${newClips.length === 2 ? "" : "s"}`, {
   785→   785→        description: `Clip shortened by ${removedTotal.toFixed(1)}s.`,
   786→   786→      });
   787→   787→    } finally {
   788→   788→      setBusy(false);
   789→   789→    }
   790→   790→  };
   791→   791→
   792→   792→  return (
   793→   793→    <div className="space-y-2">
   794→   794→      <div className="rounded-xl border border-white/8 bg-white/[0.02] p-3">
   795→   795→        <div className="flex items-center justify-between gap-2">
   796→   796→          <div>
   797→   797→            <p className="text-[11px] font-semibold text-zinc-100">Silence removal</p>
   798→   798→            <p className="mt-0.5 text-[10px] leading-relaxed text-zinc-500">
   799→   799→              Detects quiet gaps in the selected clip&apos;s waveform and tightens them automatically.
   800→   800→            </p>
   801→   801→          </div>
